🎯 PRÓXIMOS PASOS (MANUAL)

  1. INSTALAR DEPENDENCIAS (10 minutos)
     ──────────────────────────────────
     # Frontend
     cd frontend && npm install && cd ..

     # Landing
     cd landing && npm install && cd ..

     # Backend (crear venv)
     cd backend
     python -m venv venv
     source venv/bin/activate  # Windows: venv\Scripts\activate
     pip install -r requirements.txt
     cd ..

  2. CONFIGURAR VARIABLES DE ENTORNO (2 minutos)
     ────────────────────────────────────────────
     cp .env.example .env
     # Editar .env si es necesario (default values OK)

  3. LEVANTAR SERVICIOS CON DOCKER (1 minuto)
     ──────────────────────────────────────────
     docker-compose up -d

  4. CREAR SUPERUSUARIO DJANGO (2 minutos)
     ──────────────────────────────────────
     docker-compose exec backend python manage.py createsuperuser

  5. CORRER MIGRACIONES (1 minuto)
     ────────────────────────────────
     docker-compose exec backend python manage.py migrate

  6. ACCEDER A SERVICIOS (10 segundos)
     ────────────────────────────────
     Frontend:     http://localhost:5173
     Backend API:  http://localhost:8000
     Landing:      http://localhost:3000
     Admin Django: http://localhost:8000/admin

  TIEMPO TOTAL: ~16 minutos
